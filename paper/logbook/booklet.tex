% A small fillable box, used for dice, modifier check, score digits.
\newcommand{\fillbox}[1][0.22in]{%
  \tikz[baseline=-0.45ex]\draw[line width=0.4pt] (0,0) rectangle (#1,0.22in);}

% A horizontal rule the operator writes on.
\newcommand{\writeline}[1][\linewidth]{%
  \rule[-2pt]{#1}{0.4pt}}

% Day block: day name, modality, RELAY morning header, fillables.
% args: 1=DAY (e.g. MON)  2=MYSTERY  3=MODALITY  4=RELAY HEADER TEXT
%       5=show-scores flag (Y|N)
\newcommand{\dayblock}[5]{%
  \par\smallskip
  {\sffamily\bfseries\small\MakeUppercase{#1} \textperiodcentered{} #2 \textperiodcentered{} #3}\par
  \nopagebreak
  {\footnotesize\itshape RELAY: #4}\par
  \nopagebreak
  \footnotesize
  Dice \fillbox\quad Mod \fillbox\quad Work:~\writeline[2.7in]\par
  \ifthenelse{\equal{#5}{Y}}{%
    Scores: PU \fillbox \fillbox \fillbox \quad
            SU \fillbox \fillbox \fillbox \quad
            Plank \fillbox:\fillbox \fillbox \par
  }{}%
  Notes:~\writeline\par
}

% A bead spread is two pages: LEFT page (header+QR+load+Mon-Tue-Wed)
% and RIGHT page (Thu-Fri-Sat + Sun Sanctum band).
% csvsimple feeds us \beadnum, \phaselabel, \loadmult, \loadnote, etc.
%
% Bead spread is emitted by \beadspread, called inside a \csvreader loop.
\newcommand{\beadspread}{%
  % ---------------- LEFT PAGE ----------------
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north west,inner sep=0pt] at ([yshift=-0.4in,xshift=0.45in]current page.north west) {%
      \qrcode[height=0.85in]{RY|v1|Y\YearNumber|D\DecadeNumber|B\beadnum|P\phaselabel|M\DecadeMystery|TLog}%
    };
  \end{tikzpicture}%
  \vspace*{0.05in}
  \hspace*{1.0in}%
  \begin{minipage}[t]{3.35in}
    \footnotesize\sffamily\bfseries
    YEAR \YearNumber{} \textperiodcentered{} DECADE \DecadeNumber{} \textperiodcentered{}
    BEAD \beadnum{} \textperiodcentered{} \MakeUppercase{\DecadeMystery}\par
    PHASE: \MakeUppercase{\phaselabel}\par
    \mdseries\itshape WEEK OF \rule[-2pt]{0.45in}{0.3pt} / \rule[-2pt]{0.45in}{0.3pt} / \rule[-2pt]{0.45in}{0.3pt}\par
  \end{minipage}\par
  \vspace{0.10in}
  \hrule height 0.4pt
  \vspace{4pt}
  {\sffamily\bfseries\footnotesize LOAD TARGETS \textnormal{(phase \phaselabel, \loadmult{}):}}\par
  {\footnotesize \loadnote\ Targets: PU 5x\rule[-2pt]{0.25in}{0.3pt}~ SU 5x\rule[-2pt]{0.25in}{0.3pt}~ Plank 5x\rule[-2pt]{0.35in}{0.3pt}\,"~ Run \rule[-2pt]{0.35in}{0.3pt}}\par
  \vspace{4pt}\hrule height 0.4pt
  %
  \dayblock{Mon}{Joyful}{Mobility}{\relaymon}{N}
  \dayblock{Tue}{Sorrowful}{Endurance}{\relaytue}{Y}
  \dayblock{Wed}{Glorious}{Strength}{\relaywed}{Y}
  \newpage
  %
  % ---------------- RIGHT PAGE ----------------
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north east,inner sep=0pt,font=\sffamily\tiny] at
      ([yshift=-0.35in,xshift=-0.45in]current page.north east)
      {BEAD \beadnum{} / 10};
  \end{tikzpicture}%
  \vspace*{0.05in}
  \dayblock{Thu}{Luminous}{Skill}{\relaythu}{Y}
  \dayblock{Fri}{Sorrowful}{Endurance}{\relayfri}{Y}
  \dayblock{Sat}{Joyful}{Recovery}{\relaysat}{N}
  \vfill
  \hrule height 0.4pt
  \vspace{4pt}
  {\sffamily\bfseries\footnotesize SUN \textperiodcentered{} SANCTUM}\par
  {\footnotesize
  Bead closed \fillbox\quad Weight \rule[-2pt]{0.7in}{0.3pt}\,lb \quad
  Sleep avg \rule[-2pt]{0.5in}{0.3pt}\,h\par
  RELAY post-bead:~\writeline\par
  \writeline\par
  Operator note:~\writeline\par
  \writeline\par}
  \newpage
}
